cmake_minimum_required(VERSION 3.20)
project(th06_dos LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(NO_SDL "Build without SDL" ON)

# =========================================================
# Sources
# =========================================================

set(TH06_GAME_SOURCES
    ../../src/main.cpp

    ../../src/AnmManager.cpp
    ../../src/AsciiManager.cpp
    ../../src/BombData.cpp
    ../../src/BulletData.cpp
    ../../src/BulletManager.cpp
    ../../src/Chain.cpp

    ../../src/EclManager.cpp
    ../../src/EffectManager.cpp
    ../../src/Ending.cpp
    ../../src/EnemyEclInstr.cpp
    ../../src/EnemyManager.cpp
    ../../src/FileSystem.cpp
    ../../src/GameErrorContext.cpp
    ../../src/GameManager.cpp
    ../../src/GamePaths.cpp
    ../../src/GameWindow.cpp
    ../../src/Gui.cpp
    ../../src/ItemManager.cpp
    ../../src/MainMenu.cpp
    ../../src/MusicRoom.cpp
    ../../src/Player.cpp
    ../../src/ReplayManager.cpp
    ../../src/ResultScreen.cpp
    ../../src/Rng.cpp
    ../../src/ScreenEffect.cpp
    ../../src/Stage.cpp
    ../../src/Supervisor.cpp
    ../../src/TextHelper.cpp
    ../../src/utils.cpp
    ../../src/ZunMath.cpp
    ../../src/ZunTimer.cpp

    ../../src/thirdparty/sjis_converter.cpp
    ../../src/midi/MidiDefault.cpp

    # Software renderer only
    ../../src/graphics/Software2Ddos.cpp

    ../../src/pbg3/FileAbstraction.cpp
    ../../src/pbg3/IPbg3Parser.cpp
    ../../src/pbg3/Pbg3Archive.cpp
    ../../src/pbg3/Pbg3Parser.cpp
)

# =========================================================
# Platform backend
# =========================================================

if(NO_SDL)

    list(APPEND TH06_GAME_SOURCES
        ../../src/controller/EmptyController.cpp
        ../../src/sound/EmptyMidiOutput.cpp
        ../../src/sound/EmptySoundPlayer.cpp
    )

else()

    message(FATAL_ERROR "This configuration is intended for NO_SDL builds only.")

endif()

# =========================================================
# Executable
# =========================================================

add_executable(th06 ${TH06_GAME_SOURCES})

# =========================================================
# Definitions
# =========================================================

target_compile_definitions(th06 PRIVATE
    NO_SDL
    USE_DOS
    TH06_SOFTWARE_RENDER
)

# =========================================================
# Includes
# =========================================================

target_include_directories(th06 PRIVATE
    ${CMAKE_SOURCE_DIR}/../../src
    ${CMAKE_SOURCE_DIR}/../../build/autogenerated
)

# =========================================================
# Compiler flags
# =========================================================

if(MSVC)

    target_compile_options(th06 PRIVATE
        /utf-8
    )

else()

    target_compile_options(th06 PRIVATE
        -fpermissive
    )

endif()

# =========================================================
# i18n generation
# =========================================================

set(I18N_TPL "${CMAKE_SOURCE_DIR}/../../src/i18n.tpl")
set(I18N_HPP "${CMAKE_SOURCE_DIR}/../../build/autogenerated/i18n.hpp")

find_program(TH06_PYTHON NAMES python3 python REQUIRED)

add_custom_command(
    OUTPUT "${I18N_HPP}"

    COMMAND ${CMAKE_COMMAND} -E make_directory
        "${CMAKE_SOURCE_DIR}/../../build/autogenerated"

    COMMAND ${TH06_PYTHON}
        "${CMAKE_SOURCE_DIR}/../../scripts/gen_i18n.py"
        "${I18N_TPL}"
        "${I18N_HPP}"
        "utf-8"

    DEPENDS
        "${I18N_TPL}"
        "${CMAKE_SOURCE_DIR}/../../scripts/gen_i18n.py"

    COMMENT "Generating i18n.hpp"
)

add_custom_target(generate_i18n DEPENDS "${I18N_HPP}")

add_dependencies(th06 generate_i18n)

# =========================================================
# Font copy
# =========================================================

set(FONT_SRC "${CMAKE_SOURCE_DIR}/../../th06.ttf")

if(EXISTS "${FONT_SRC}")

    add_custom_command(TARGET th06 POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${FONT_SRC}"
            "$<TARGET_FILE_DIR:th06>/th06.ttf"
        COMMENT "Copying font"
    )

endif()

# =========================================================
# Build summary
# =========================================================

message(STATUS "Building TH06")
message(STATUS "Renderer : Software")
message(STATUS "SDL      : Disabled")
message(STATUS "NO_SDL   : Enabled")
